## 第二部分：核心流程实现

### 2.1 创建本地化任务（step1_localize）

**业务背景**：
- 用户选择一篇小说，后端需要调用 AI 能力层进行本地化改写
- 本地化是异步操作，后端创建 Job 后轮询等待

**实现步骤**：

```python
def submit_localization_job(project_id, novel_text, prompt_config):
    """
    Args:
        project_id: str, 业务项目 ID
        novel_text: str, 小说原文（UTF-8，最大 1MB）
        prompt_config: dict, {system, user, work_note}
    Returns:
        job_id: str, AI 能力层返回的 job_id
    """
    
    # 1. 构造请求体
    request_body = {
        'client_request_id': f"project_{project_id}_step1_{uuid.uuid4()}",
        'job_type': 'novel_localization.step1_localize',
        'model_id': 'gpt-4o-mini',  # 来自 /models 返回的 id
        'input': {
            'type': 'text',
            'content': novel_text,
            'content_hash': f"sha256:{hashlib.sha256(novel_text.encode()).hexdigest()}"
        },
        'output': {
            'type': 'oss_prefix',
            'oss_bucket': 'your-output-bucket',
            'oss_prefix': f"novel-localization/projects/{project_id}/",
            'oss_region': 'cn-hangzhou'
        },
        'callback': {
            'url': f"{YOUR_BACKEND_HOST}/internal/ai-callbacks/novel-localization",
            'events': ['job.succeeded', 'job.failed']
        },
        'prompt': {
            'blocks': [
                {
                    'key': 'system',
                    'role': 'system',
                    'content': prompt_config['system']
                },
                {
                    'key': 'user',
                    'role': 'user',
                    'content': prompt_config['user']
                },
                {
                    'key': 'work_note',
                    'role': 'user',
                    'content': prompt_config.get('work_note', '')
                }
            ]
        },
        'metadata': {
            'external_project_ref': project_id,
            'external_job_ref': f"step1_{uuid.uuid4()}"
        }
    }
    
    # 2. 调用 POST /jobs
    response = requests.post(
        f"{AI_SERVICE_HOST}/api/v1/novel-localization-ai/jobs",
        json=request_body,
        headers={'Authorization': f'Bearer {API_KEY}'},
        timeout=30
    )
    
    # 3. 错误处理
    if response.status_code != 202:
        error = response.json()['error']
        raise Exception(f"创建 Job 失败: {error['code']} - {error['message']}")
    
    # 4. 保存任务记录到业务库
    job_id = response.json()['job_id']
    DB.save_ai_job({
        'job_id': job_id,
        'project_id': project_id,
        'job_type': 'step1_localize',
        'status': 'queued',
        'created_at': datetime.now()
    })
    
    return job_id
```

**检查清单**：
- [ ] 能成功创建 step1 Job，返回 202 + job_id
- [ ] content_hash 格式正确（sha256:lowercase-hex）
- [ ] metadata 中的字段不超过 8 KB

### 2.2 轮询任务状态

**实现步骤**：

```python
import time
from enum import Enum

class JobStatus(Enum):
    QUEUED = 'queued'
    RUNNING = 'running'
    SUCCEEDED = 'succeeded'
    FAILED = 'failed'
    CANCELED = 'canceled'

def poll_job_until_complete(job_id, max_wait_seconds=1800):
    """
    轮询任务直到完成或超时。
    
    Args:
        job_id: str
        max_wait_seconds: int, 最长等待时间（默认 30 分钟）
    
    Returns:
        job_result: dict, 任务的最终响应
    """
    
    start_time = time.time()
    poll_interval = 2  # 初始轮询间隔 2 秒
    
    while True:
        # 1. 检查超时
        elapsed = time.time() - start_time
        if elapsed > max_wait_seconds:
            raise TimeoutError(f"Job {job_id} 执行超时 {max_wait_seconds}s")
        
        # 2. 查询任务状态
        try:
            response = requests.get(
                f"{AI_SERVICE_HOST}/api/v1/novel-localization-ai/jobs/{job_id}",
                headers={'Authorization': f'Bearer {API_KEY}'},
                timeout=10
            )
        except requests.RequestException as e:
            # 请求失败，使用指数退避
            poll_interval = min(poll_interval * 2, 60)
            logger.warning(f"查询 Job {job_id} 失败: {e}，{poll_interval}s 后重试")
            time.sleep(poll_interval)
            continue
        
        if response.status_code != 200:
            logger.error(f"查询失败: {response.status_code} {response.text}")
            raise Exception(f"查询 Job 失败: {response.status_code}")
        
        job_result = response.json()
        status = job_result['status']
        progress = job_result['progress_percent']
        
        # 3. 更新业务库进度
        DB.update_ai_job_progress(job_id, {
            'status': status,
            'progress_percent': progress,
            'progress_text': job_result.get('progress_text')
        })
        
        # 4. 判断是否完成
        if status == JobStatus.SUCCEEDED.value:
            logger.info(f"Job {job_id} 完成，进度: {progress}%")
            return job_result
        
        elif status == JobStatus.FAILED.value:
            error = job_result.get('error', {})
            raise Exception(
                f"Job {job_id} 执行失败: {error.get('code')} - {error.get('message')}"
            )
        
        elif status == JobStatus.CANCELED.value:
            raise Exception(f"Job {job_id} 已被取消")
        
        # 5. 还在执行中，等待后重新轮询
        logger.debug(f"Job {job_id} 进度: {progress}%, 状态: {status}")
        time.sleep(poll_interval)
```

**检查清单**：
- [ ] 能成功轮询任务状态
- [ ] 正确处理超时情况
- [ ] 连续请求失败时做指数退避
- [ ] 能识别成功、失败、取消三种终态
- [ ] 进度信息正确更新到业务库

### 2.3 处理 step1 结果

**业务背景**：
- step1 完成后返回本地化文本和其他产物
- 大文本（localized_text）存在 OSS 中，需要读取

**实现步骤**：

```python
def process_step1_result(job_result):
    """
    处理 step1 的执行结果。
    
    Args:
        job_result: dict, GET /jobs/{job_id} 的响应
    
    Returns:
        processed: dict, {localized_text, notes, project_memory}
    """
    
    if job_result['status'] != 'succeeded':
        raise Exception("Job 未成功完成")
    
    result_payload = job_result['result']
    artifacts = result_payload['artifacts']
    signals = result_payload.get('signals', {})
    
    processed = {}
    
    # 1. 提取本地化文本（存在 OSS 中）
    localized_text_artifact = next(
        (a for a in artifacts if a['key'] == 'localized_text'),
        None
    )
    
    if localized_text_artifact:
        if localized_text_artifact.get('storage') == 'oss_object':
            # 读取 OSS 对象
            oss_text = read_from_oss(
                bucket=localized_text_artifact['oss_bucket'],
                key=localized_text_artifact['oss_key'],
                region=localized_text_artifact['oss_region']
            )
            processed['localized_text'] = oss_text
            processed['localized_text_oss'] = {
                'bucket': localized_text_artifact['oss_bucket'],
                'key': localized_text_artifact['oss_key'],
                'hash': localized_text_artifact.get('content_hash'),
                'size_bytes': localized_text_artifact.get('content_size_bytes')
            }
        else:
            # 小文本直接返回
            processed['localized_text'] = localized_text_artifact.get('content', '')
    
    # 2. 提取工作注释
    notes_artifact = next(
        (a for a in artifacts if a['key'] == 'notes'),
        None
    )
    processed['notes'] = notes_artifact.get('content', '') if notes_artifact else ''
    
    # 3. 提取项目映射表（仅 P5 分块模式下存在）
    memory_artifact = next(
        (a for a in artifacts if a['key'] == 'project_memory'),
        None
    )
    if memory_artifact:
        processed['project_memory'] = memory_artifact.get('content')
    
    # 4. 检查信号
    if signals.get('project_memory_frozen'):
        processed['has_memory'] = True
    
    # 5. 保存到业务库
    DB.save_step1_result({
        'job_id': job_result['job_id'],
        'localized_text': processed.get('localized_text'),
        'localized_text_oss_key': processed.get('localized_text_oss', {}).get('key'),
        'notes': processed.get('notes'),
        'project_memory': processed.get('project_memory'),
        'completed_at': datetime.now()
    })
    
    return processed
```

- [ ] 能读取 OSS 中的本地化文本
- [ ] 能提取 notes 和 project_memory（如有）
- [ ] 信号 project_memory_frozen 正确识别
- [ ] 结果保存到业务库

## 第三部分：错误处理和边界情况

### 3.1 常见错误码处理

| 错误码 | 原因 | 后端应对 |
|---|---|---|
| `MODEL_NOT_AVAILABLE` | 指定的 model_id 不可用 | 从 `/models` 重新获取可用模型列表，提示用户 |
| `INVALID_JOB_TYPE` | job_type 不支持 | 检查请求中的 job_type 是否来自 `/prompt-templates` |
| `INPUT_TOO_LARGE` | 输入超过大小限制（1MB for text, 5MB for oss_object） | 提示用户文本过长，建议分割后多次提交 |
| `INPUT_HASH_MISMATCH` | content_hash 校验失败 | 重新计算文本的 SHA256 hash，检查是否被修改 |
| `OSS_FETCH_FAILED` | 读取 OSS 输入失败 | 检查 OSS bucket/key/region 是否正确 |
| `OSS_WRITE_FAILED` | 写入 OSS 输出失败 | 检查 AI 能力层的 OSS 凭证和 bucket 权限 |
| `JOB_TIMEOUT` | Job 排队或执行超时 | 重新提交任务（可用 client_request_id 幂等） |
| `MODEL_CALL_FAILED` | 模型调用失败 | 重试（最多 3 次），或人工介入 |

```python
def handle_job_error(job_result):
    """处理 Job 执行错误。"""
    
    error = job_result['error']
    code = error['code']
```

### 3.2 Callback 签名验证

**业务背景**：
- 当 Job 完成时，AI 能力层会 POST callback.url
- 后端必须校验签名，防止伪造

**实现步骤**：

```python
import hmac
import hashlib
from datetime import datetime, timedelta

def verify_callback_signature(request_headers, request_body, callback_signature_secret):
    """
    验证 callback 签名。
    
    Args:
        request_headers: dict, 包含 X-AI-Service-Timestamp 和 X-AI-Service-Signature
        request_body: bytes, 原始请求体
        callback_signature_secret: str, 签名密钥
    
    Returns:
        bool, 签名是否有效
    """
    
    timestamp = request_headers.get('X-AI-Service-Timestamp')
    signature = request_headers.get('X-AI-Service-Signature')
    
    if not timestamp or not signature:
        logger.error("缺少签名头")
        return False
    
    # 1. 验证 timestamp 不超过 5 分钟（防重放）
    try:
        callback_time = datetime.fromisoformat(timestamp.replace('Z', '+00:00'))
        current_time = datetime.utcnow()
        time_diff = (current_time - callback_time).total_seconds()
        
        if abs(time_diff) > 300:  # 5 分钟
            logger.error(f"Timestamp 过期: {timestamp}")
            return False
    except ValueError:
        logger.error(f"Timestamp 格式错误: {timestamp}")
        return False
    
    # 2. 计算期望的签名
    # 签名原文：timestamp + "." + request_body
    signature_raw = f"{timestamp}.{request_body.decode() if isinstance(request_body, bytes) else request_body}"
    
    expected_signature = "sha256=" + hmac.new(
        callback_signature_secret.encode(),
        signature_raw.encode(),
        hashlib.sha256
    ).hexdigest()
    
    # 3. 对比签名（使用恒定时间对比，防时序攻击）
    is_valid = hmac.compare_digest(signature, expected_signature)
    
    if not is_valid:
        logger.error(f"签名验证失败: {signature} vs {expected_signature}")
    
    return is_valid

# Flask 示例
@app.post('/internal/ai-callbacks/novel-localization')
def handle_ai_callback():
    """处理 AI 能力层的 callback 通知。"""
    
    # 1. 验证签名
    if not verify_callback_signature(
        request.headers,
        request.get_data(),
        os.getenv('AI_CALLBACK_SIGNATURE_SECRET')
    ):
        return {'error': 'INVALID_SIGNATURE'}, 401
    
    # 2. 解析 callback 数据
    callback_data = request.json
    job_id = callback_data['job_id']
    job_type = callback_data['job_type']
    status = callback_data['status']
    event = callback_data['event']
    
    # 3. 幂等去重（业务后端需维护去重表）
    dedup_key = f"{job_id}#{event}"
    if DB.has_processed_callback(dedup_key):
        logger.info(f"Callback 已处理过: {dedup_key}")
        return {}, 200
    
    # 4. 处理不同的事件
    if event == 'job.succeeded':
        handle_job_succeeded(callback_data)
    elif event == 'job.failed':
        handle_job_failed(callback_data)
    
    # 5. 标记已处理
    DB.mark_callback_processed(dedup_key)
    
    # 6. 返回 2xx 表示成功接收
    return {}, 200
```

### 3.3 超时和重试策略

| 场景 | 超时 | 重试策略 |
|---|---|---|
| POST /jobs 请求 | 30 秒 | 不重试（调用方处理） |
| GET /jobs/{job_id} 轮询 | 10 秒 | 连续失败时指数退避（2s → 4s → 8s → 16s） |
| OSS 读写 | 60 秒 | 3 次重试，间隔 2s/5s/10s |
| Callback POST | 30 秒 | AI 能力层负责重试 |

```python
def poll_with_exponential_backoff(job_id, max_retries=10):
    """
    带指数退避的轮询。
    """
    
    poll_interval = 2
    consecutive_failures = 0
    
    while consecutive_failures < max_retries:
        try:
            response = requests.get(
                f"{AI_SERVICE_HOST}/api/v1/novel-localization-ai/jobs/{job_id}",
                headers={'Authorization': f'Bearer {API_KEY}'},
                timeout=10
            )
            
            if response.status_code == 200:
                consecutive_failures = 0  # 重置计数
                return response.json()
            else:
                consecutive_failures += 1
                poll_interval = min(poll_interval * 2, 60)  # 上限 60 秒
        
        except requests.RequestException:
            consecutive_failures += 1
            poll_interval = min(poll_interval * 2, 60)
        
        logger.warning(f"轮询失败 {consecutive_failures} 次，{poll_interval}s 后重试")
        time.sleep(poll_interval)
    
    raise TimeoutError(f"轮询 {job_id} 超过最大重试次数")
```

- [ ] 轮询间隔从 2 秒开始，指数增长，上限 60 秒
- [ ] 轮询总超时 30 分钟（与 AI 能力层的执行超时一致）
- [ ] 临时失败自动重试，永久失败快速失败
